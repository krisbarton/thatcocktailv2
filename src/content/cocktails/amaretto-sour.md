ingredients:
  - { item: '<a href="/alcohol-recommendations/#amaretto">amaretto</a>', ml: 60, oz: 2 }
  - { item: lemon juice (freshly squeezed), ml: 30, oz: 1 }
  - { item: '<a href="/alcohol-recommendations/#angostura-bitters">Angostura bitters</a>', count: 1, one: dash, many: dashes }
  - { item: egg white, ml: 15, oz: 0.5 }

method:
  - 'Add your <a href="/alcohol-recommendations/#amaretto">amaretto</a>, lemon juice, <a href="/alcohol-recommendations/#angostura-bitters">Angostura bitters</a>, and egg whites to a <a href="/cocktail-equipment/#cocktail-shaker">cocktail shaker</a>.'
  - Shake for 15 seconds.
  - Add ice and then shake again until well chilled.
  - 'Add ice to an <a href="/cocktail-glassware/#old-fashioned-glasses">Old Fashioned glass</a>.'
  - 'Use a <a href="/cocktail-equipment/#hawthorne-strainer">Hawthorne strainer</a> to strain your mix into the glass.'
  - Garnish with maraschino cherries.
