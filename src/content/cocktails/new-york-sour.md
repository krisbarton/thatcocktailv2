ingredients:
  - { item: '<a href="/alcohol-recommendations/#bourbon">bourbon</a>', ml: 60, oz: 2 }
  - { item: lemon juice, ml: 30, oz: 1 }
  - { item: red wine (chilled), ml: 15, oz: 0.5 }
  - { item: simple syrup, ml: 15, oz: 0.5 }
  - { item: '<a href="/alcohol-recommendations/#angostura-bitters">Angostura bitters</a>', count: 1, one: dash, many: dashes }
  - { item: egg whites (or aquafaba), ml: 15, oz: 0.5 }
  - { item: slice of lemon, garnish: true, note: to garnish }
method:
  - 'Add the <a href="/alcohol-recommendations/#bourbon">bourbon</a>, <a href="/alcohol-recommendations/#angostura-bitters">Angostura bitters</a>, lemon juice, simple syrup, and egg whites (or aquafaba) to your <a href="/cocktail-equipment/#cocktail-shaker">cocktail shaker</a> and shake well.'
  - 'Add ice to the <a href="/cocktail-equipment/#cocktail-shaker">cocktail shaker</a> and shake again until the mix is well-chilled.'
  - 'Strain into an <a href="/cocktail-glassware/#old-fashioned-glasses">Old Fashioned glass</a> over ice.'
  - Pour the red wine over the back of a bar spoon so it’s distributed evenly and floats on top of the mix.
  - Garnish with a slice of lemon.
